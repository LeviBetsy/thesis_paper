% Reduction of the flattened point cloud to a 1D angular range scan.
%
% Top-down view: the camera sits at the origin looking up (+y). The field of
% view (70 deg here, schematic) is split into 4 slices -- the real scan uses
% 16, fewer are drawn for legibility. Each ray goes to the nearest point in
% its slice; slices with no point inside r_max (radius 5) end on the r_max
% arc with a hollow circle, matching the convention of the ray-cast figure
% in chapter 5. Nearest-point coordinates were computed exactly from the
% point list below.
%
% Pure vector TikZ -- solid fills and plain paths only. Do NOT introduce
% opacity, shadings or fadings here: the Makefile builds via
% latex -> dvips -> dvipdf, where those features are unreliable.
\begin{tikzpicture}[>=Latex, line join=round, font=\small, scale=0.85]

  \colorlet{raycol}{blue!70!black}

  % ---- r_max arc and slice boundaries ---------------------------------
  \draw[black!45] (55:5) arc (55:125:5);
  \foreach \a in {55,72.5,90,107.5,125}
    \draw[black!35,dashed] (0,0) -- (\a:5);
  \draw[black!60] (0,0) -- (55:5.6);
  \draw[black!60] (0,0) -- (125:5.6)
    node[anchor=south east,black!60] {field of view};

  % ---- flattened obstacle points --------------------------------------
  % wall, left
  \foreach \p in {(-2.70,3.17),(-2.53,3.21),(-2.36,3.18),(-2.19,3.21),(-2.02,3.22),
                  (-1.85,3.15),(-1.68,3.14),(-1.51,3.24),(-1.34,3.17),(-1.17,3.17),
                  (-1.00,3.26),(-0.83,3.20),(-0.66,3.24),(-0.49,3.20)}
    \fill[black!40] \p circle (1.3pt);
  % box, front face and left side
  \foreach \p in {(0.90,2.21),(1.06,2.17),(1.22,2.21),(1.38,2.24),(1.54,2.20),
                  (1.70,2.22),(0.91,2.36),(0.87,2.52),(0.92,2.68),(0.91,2.84)}
    \fill[black!40] \p circle (1.3pt);
  % far wall, beyond r_max
  \foreach \p in {(-1.80,5.58),(-1.55,5.54),(-1.30,5.64),(-1.05,5.60),(-0.80,5.63),
                  (-0.55,5.65),(-0.30,5.63),(-0.05,5.65),(0.20,5.59),(0.45,5.64),
                  (0.70,5.59),(0.95,5.65),(1.20,5.65),(1.45,5.55),(1.70,5.56),
                  (1.95,5.57),(2.20,5.66),(2.45,5.59)}
    \fill[black!40] \p circle (1.3pt);

  % ---- rays: nearest point per slice, or r_max if empty -----------------
  \draw[raycol,thick] (0,0) -- (-1.17,3.17); \fill[raycol] (-1.17,3.17) circle (2pt);
  \draw[raycol,thick] (0,0) -- (-0.49,3.20); \fill[raycol] (-0.49,3.20) circle (2pt);
  \draw[raycol,thick] (0,0) -- (81.25:5);
  \draw[raycol,thick,fill=white] (81.25:5) circle (2.4pt);
  \draw[raycol,thick] (0,0) -- (0.90,2.21); \fill[raycol] (0.90,2.21) circle (2pt);

  % ---- labels ------------------------------------------------------------
  \node[anchor=west,align=left] at (2.6,1.2) {nearest point\\in slice};
  \draw[black!60] (2.55,1.35) -- (0.98,2.14);
  \path (0,0) -- (81.25:5)
    node[pos=0.72,sloped,anchor=south,raycol] {$r_{\max}$};

  % ---- rover and camera --------------------------------------------------
  \draw[black!75,fill=black!10] (-0.7,-1.6) rectangle (0.7,-0.2);
  \node at (0,-0.9) {rover};
  \fill[black!70] (0,0) -- (0.22,-0.2) -- (-0.22,-0.2) -- cycle;
  \node[anchor=east] at (-0.3,-0.05) {camera};

\end{tikzpicture}
